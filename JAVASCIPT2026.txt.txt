import React, { useState, useEffect } from 'react';
import { View, FlatList, Text, ActivityIndicator, StyleSheet, Dimensions } from 'react-native';

export default function App() {
  const [user, setUser] = useState([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetch('https://jsonplaceholder.typicode.com/users')
      .then((res) => res.json())
      .then((data) => {
        setUser(data);
        setLoading(false);
      })
      .catch((errr) => {
        console.error("Error fetching data", errr);
        setLoading(false);
      });
  }, []);

  if (loading) {
    return (
      <View style={styles.centerContainer}>
        <ActivityIndicator size="large" color="#0000ff" />
        <Text style={{ marginTop: 10 }}>Loading...</Text>
      </View>
    );
  }

  return (
    <View style={styles.container}>
      <FlatList
        data={user}
        keyExtractor={(item) => item.id.toString()}
        // 1. contentContainerStyle centers the list items vertically if they don't fill the screen
        contentContainerStyle={styles.listContent}
        renderItem={({ item }) => (
          // 2. Each text item is wrapped in a View to center it horizontally
          <View style={styles.itemContainer}>
            <Text style={styles.itemText}>{item.name}</Text>
          </View>
        )}
      />
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    backgroundColor: '#fff',
  },
  centerContainer: {
    flex: 1,
    justifyContent: 'center',
    alignItems: 'center',
  },
  listContent: {
    flexGrow: 1,                  // Ensures the list takes full height to allow vertical centering
    justifyContent: 'center',     // Centers items vertically on the screen
    alignItems: 'center',         // Centers items horizontally on the screen
  },
  itemContainer: {
    width: Dimensions.get('window').width, // Makes the container take full width of the screen
    alignItems: 'center',         // Forces the Text component inside to stay dead center
    paddingVertical: 10,
  },
  itemText: {
    fontSize: 18,
    textAlign: 'center',          // Centers the text alignment itself
  }
});
